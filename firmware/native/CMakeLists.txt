cmake_minimum_required(VERSION 3.16)
project(openbyddisplay_video_core LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(OBD_SANITIZE "Enable host address/undefined behavior sanitizers" OFF)
if(OBD_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra -Werror -pedantic)
add_library(obd_video_core core/lhf2.c)
target_include_directories(obd_video_core PUBLIC core)
add_library(obd_trial_core core/trial.c)
target_include_directories(obd_trial_core PUBLIC core)
add_library(obd_generation_core core/generation.c core/sha256.c core/trial_record.c core/profile.c
  vendor/monocypher/monocypher.c vendor/monocypher/monocypher-ed25519.c)
target_include_directories(obd_generation_core PUBLIC core vendor/monocypher)
target_link_libraries(obd_generation_core PUBLIC obd_trial_core)
add_executable(native-store-host qnx66/install_main.c qnx66/store.c qnx66/disk.c)
target_include_directories(native-store-host PRIVATE qnx66 core)
target_compile_definitions(native-store-host PRIVATE OBD_IO_HOST OBD_STORE_FAULT_TEST)
target_link_libraries(native-store-host PRIVATE obd_generation_core)
add_executable(native-updater-host qnx66/update_main.c qnx66/store.c qnx66/disk.c qnx66/network.c qnx66/io.c core/update_request.c)
target_include_directories(native-updater-host PRIVATE qnx66 core)
target_compile_definitions(native-updater-host PRIVATE OBD_IO_HOST OBD_STORE_FAULT_TEST)
target_link_libraries(native-updater-host PRIVATE obd_generation_core ${CMAKE_DL_LIBS})
add_executable(native-supervisor-host qnx66/boot_main.c qnx66/store.c qnx66/disk.c qnx66/io.c qnx66/process_wait.c)
target_include_directories(native-supervisor-host PRIVATE qnx66 core)
target_compile_definitions(native-supervisor-host PRIVATE OBD_IO_HOST OBD_SUPERVISOR_TEST)
target_link_libraries(native-supervisor-host PRIVATE obd_generation_core)
add_executable(qnx-wait-abi-tests tests/test_process_wait.c qnx66/process_wait.c)
target_include_directories(qnx-wait-abi-tests PRIVATE qnx66)
target_compile_definitions(qnx-wait-abi-tests PRIVATE waitpid=obd_wait_fixture)
add_test(NAME qnx-wait-abi COMMAND qnx-wait-abi-tests)
foreach(kind healthy unhealthy)
  add_executable(supervisor-child-${kind} tests/supervisor_child.c qnx66/guard.c qnx66/io.c)
  target_include_directories(supervisor-child-${kind} PRIVATE qnx66 core)
  target_compile_definitions(supervisor-child-${kind} PRIVATE OBD_IO_HOST)
endforeach()
target_compile_definitions(supervisor-child-unhealthy PRIVATE OBD_TEST_CHILD_UNHEALTHY)
add_executable(trial-policy-tests tests/test_trial.c)
target_link_libraries(trial-policy-tests PRIVATE obd_trial_core)
add_executable(obd-lhf2-unpack host/unpack.c)
target_link_libraries(obd-lhf2-unpack PRIVATE obd_video_core)
add_executable(obd-bvl1-receive host/receive.c)
target_link_libraries(obd-bvl1-receive PRIVATE obd_video_core)
enable_testing()
add_test(NAME native-trial-policy COMMAND trial-policy-tests)
add_library(obd_backend_policy core/usb_txn.c core/hmi_txn.c qnx66/usb_plan.c)
target_include_directories(obd_backend_policy PUBLIC core qnx66)
add_executable(backend-policy-tests tests/test_backend_txn.c)
target_link_libraries(backend-policy-tests PRIVATE obd_backend_policy)
add_test(NAME native-backend-policy COMMAND backend-policy-tests)
add_executable(service-profile-tests tests/test_profile.c core/profile.c)
target_include_directories(service-profile-tests PRIVATE core)
add_test(NAME signed-service-profile COMMAND service-profile-tests)
add_executable(baseline-tests tests/test_baseline.c core/baseline.c core/sha256.c)
target_include_directories(baseline-tests PRIVATE core)
add_test(NAME model-baseline COMMAND baseline-tests)
add_library(player-gate-object OBJECT qnx66/player.c)
target_compile_definitions(player-gate-object PRIVATE main=obd_test_player_main)
target_include_directories(player-gate-object PRIVATE core qnx66)
add_executable(player-gate-tests tests/test_player_gate.c $<TARGET_OBJECTS:player-gate-object>)
target_include_directories(player-gate-tests PRIVATE core qnx66)
add_test(NAME player-baseline-gate COMMAND player-gate-tests)
add_executable(readonly-tests tests/test_readonly.c qnx66/readonly.c)
target_compile_definitions(readonly-tests PRIVATE OBD_IO_HOST)
target_include_directories(readonly-tests PRIVATE qnx66)
add_test(NAME readonly-host-files COMMAND readonly-tests)

target_compile_definitions(readonly-qnx-tests PRIVATE open=obd_fixture_open read=obd_fixture_read
  close=obd_fixture_close lstat=obd_fixture_lstat fstat=obd_fixture_fstat)
target_include_directories(readonly-qnx-tests PRIVATE qnx66)
add_test(NAME readonly-qnx-layout COMMAND readonly-qnx-tests)
add_executable(platform-inspect-tests tests/test_inspect.c qnx66/inspect.c)
target_include_directories(platform-inspect-tests PRIVATE qnx66 core)
add_test(NAME platform-inspection COMMAND platform-inspect-tests)
add_executable(media-load-tests tests/test_media_load.c qnx66/media_load.c)
target_include_directories(media-load-tests PRIVATE qnx66 core)
target_compile_definitions(media-load-tests PRIVATE dlopen=obd_fixture_dlopen dlsym=obd_fixture_dlsym dlclose=obd_fixture_dlclose)
add_test(NAME media-load-diagnostics COMMAND media-load-tests)
add_executable(video-core-tests tests/test_lhf2.c)
target_link_libraries(video-core-tests PRIVATE obd_video_core)
add_test(NAME video-core COMMAND video-core-tests)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME native-signed-store COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_native_store.py $<TARGET_FILE:native-store-host>)
add_test(NAME native-update-transport COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_native_update.py $<TARGET_FILE:native-updater-host> $<TARGET_FILE:native-store-host>)
add_test(NAME native-supervisor COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_supervisor.py
  $<TARGET_FILE:native-store-host> $<TARGET_FILE:native-supervisor-host>
  $<TARGET_FILE:supervisor-child-healthy> $<TARGET_FILE:supervisor-child-unhealthy>)
add_library(obd_media_model SHARED qnx66/media.c core/lhf2.c qnx66/stream.c qnx66/io.c)
target_compile_definitions(obd_media_model PRIVATE OBD_IO_HOST)
target_include_directories(obd_media_model PRIVATE core qnx66)
if(OBD_SANITIZE)
  # ctypes loads this library into an ordinary Python process. The standalone C
  # core/receiver remain sanitized; this explicit ABI-double model is not.
  target_compile_options(obd_media_model PRIVATE -fno-sanitize=all)
  target_link_options(obd_media_model PRIVATE -fno-sanitize=all)
endif()
add_test(NAME media-lifecycle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_media.py $<TARGET_FILE:obd_media_model>)
add_test(NAME media-real-video COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_media_video.py $<TARGET_FILE:obd_media_model>)
add_test(NAME continuous-stream COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_stream.py $<TARGET_FILE:obd_media_model>)
